cmake_minimum_required(VERSION 3.20)
project(SpektraFilmHDR LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(UHDR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(UHDR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(UHDR_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(UHDR_BUILD_FUZZERS OFF CACHE BOOL "" FORCE)
set(UHDR_ENABLE_HEIF OFF CACHE BOOL "" FORCE)
set(UHDR_ENABLE_GLES OFF CACHE BOOL "" FORCE)
set(UHDR_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(UHDR_WRITE_XMP ON CACHE BOOL "" FORCE)
set(UHDR_WRITE_ISO ON CACHE BOOL "" FORCE)
set(UHDR_MAX_DIMENSION 65500 CACHE STRING "JPEG limit; never cap high-resolution photographs to 8192" FORCE)
add_subdirectory(upstream/libultrahdr-2.0.2)
add_library(spektrafilm_hdr SHARED bridge.cpp)
set_target_properties(spektrafilm_hdr PROPERTIES PREFIX "")
target_compile_features(spektrafilm_hdr PRIVATE cxx_std_17)
target_link_libraries(spektrafilm_hdr PRIVATE uhdr)
if(MINGW)
  target_link_options(spektrafilm_hdr PRIVATE -static-libgcc -static-libstdc++ -static)
endif()
